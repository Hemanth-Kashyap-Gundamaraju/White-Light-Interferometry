import numpy as np
import matplotlib.pyplot as plt
import pandas as pd

# Function to load heightmap from CSV
def load_heightmap(filename="heightmap.csv"):
    data = pd.read_csv(filename)
    max_x = int(data["x"].max()) + 1
    max_y = int(data["y"].max()) + 1
    Z = np.zeros((max_y, max_x))

    for _, row in data.iterrows():
        x = int(row["x"])
        y = int(row["y"])
        z = row["z"]
        Z[y, x] = z
    return Z

# Function to convert height to nm scale
def to_nm(z_surface, scale=100.0):
    return scale * z_surface

# Function to expand for broadcasting
def expand_for_broadcast(z_surface_nm, s_range):
    z_exp = z_surface_nm[:, :, np.newaxis]  # (y, x, 1)
    s_exp = s_range[np.newaxis, np.newaxis, :]  # (1, 1, s)
    return z_exp, s_exp

# Function to compute optical path difference
def compute_opd(z_exp, s_exp):
    return 2 * (z_exp - s_exp)

# Function to compute degree of coherence
def compute_gamma(OPD, Lc):
    return np.exp(-(OPD**2) / (2 * Lc**2))

# Function to compute interferogram intensity
def compute_intensity(gamma, OPD, lambda0):
    return 1 + gamma * np.cos((2 * np.pi / lambda0) * OPD)

# Function to simulate a white light interferogram
def simulate_interferogram(z_surface, lambda0=600.0, Lc=100.0, s_range=None, n_wl=41, V0=0.9):
    if s_range is None:
        s_range = np.linspace(-300, 300, 200)

    z_surface_nm = to_nm(z_surface)
    z_exp, s_exp = expand_for_broadcast(z_surface_nm, s_range)
    OPD = compute_opd(z_exp, s_exp)

    delta_lambda_fwhm = (lambda0**2) / (Lc + 1e-12)
    sigma_lambda = delta_lambda_fwhm / (2.0 * np.sqrt(2.0 * np.log(2.0)))

    wl = np.linspace(lambda0 - 2 * delta_lambda_fwhm, lambda0 + 2 * delta_lambda_fwhm, n_wl)
    S = np.exp(-0.5 * ((wl - lambda0) / sigma_lambda)**2)
    S /= S.sum()

    I = np.zeros_like(OPD, dtype=float)
    for w, weight in zip(wl, S):
        phase = (2.0 * np.pi / w) * OPD
        I += weight * (1.0 + V0 * np.cos(phase))

    return I, s_range

# Load heightmap
try:
    heightmap = load_heightmap("heightmap.csv")
    print("Loaded heightmap.csv")
except FileNotFoundError:
    print("Heightmap file not found. Please ensure 'heightmap.csv' is in the working directory.")
    heightmap = np.zeros((64, 64))  # Fallback to a zero array

# Simulation parameters
lambda0 = 600.0  # central wavelength (nm)
Lc = 100.0       # coherence length (nm)
s_range = np.linspace(-300, 300, 200)  # reference mirror scan (nm)

# Run simulation
I, s_range = simulate_interferogram(heightmap, lambda0, Lc, s_range)

# Visualization
fig = plt.figure(figsize=(6, 9))

# Intensity pattern
ax1 = fig.add_subplot(311)
ax1.imshow(I[:, :, len(s_range) // 2], cmap='jet')
ax1.set_title('Intensity Pattern')
ax1.axis('off')

# Annotations
s1 = "Central Wavelength: 600 nm"
s2 = "Coherence Length: 100 nm"
ax2 = fig.add_subplot(312)
ax2.axis('off')
ax2.text(0.0, 0.6, s1, fontsize=12, fontweight='bold')
ax2.text(0.0, 0.5, s2, fontsize=12)

# Mean intensity plot
I_mean = I.mean(axis=(0, 1))  # average over y,x
ax3 = fig.add_subplot(313)
ax3.plot(s_range, I_mean)
ax3.set_xlabel('Reference Mirror Position s (nm)')
ax3.set_ylabel('Mean Intensity')
ax3.set_title('Mean Intensity vs. Mirror Position')
ax3.grid(True)

plt.tight_layout()
plt.show()
